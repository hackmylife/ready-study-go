package koan

import (
	"strings"
)

func Join(parts []string, sep string) string {
	if len(parts) == 0 {
		return ""
	}
	size := len(sep) * len(parts)
	for _, part := range parts {
		size += len(part)
	}
	var b strings.Builder
	b.Grow(size)
	for _, part := range parts {
		b.WriteString(part)
		b.WriteString(sep)
	}
	return b.String()
}
